44	async function authorizedFetch(input: string, init: RequestInit = {}, opts: { includeAdminToken?: boolean } = {}) {
45	  const accessToken = await getValidAccessToken();
46	  if (!accessToken) throw new Error("Not authenticated");
47	  const headers = new Headers(init.headers || {});
48	  headers.set("Authorization", `Bearer ${accessToken}`);
49	  if (!headers.has("accept")) headers.set("accept", "application/json");
50	  if (opts.includeAdminToken) headers.set("x-admin-token", DEV_TOKEN || "");
51	  return fetch(input, { ...init, headers });
52	}

54	async function fetchProducts({ q, category, status, ratingMin, page, limit, sort, dir }: FetchParams) {
55	  const url = new URL("/api/ecom-products", window.location.origin);
56	  if (q) url.searchParams.set("q", q);
57	  if (category && category !== "all") url.searchParams.set("category", category);
58	  if (status && status !== "all") url.searchParams.set("status", status);
59	  if (ratingMin) url.searchParams.set("min_rating", String(ratingMin));
60	  url.searchParams.set("page", String(page));
61	  url.searchParams.set("limit", String(limit));
62	  url.searchParams.set("sort", sort);
63	  url.searchParams.set("dir", dir);
64	
65	  const res = await authorizedFetch(url.toString());
66	  if (!res.ok) throw new Error(await res.text());
67	  const json = await res.json();
68	  return {
69	    items: (json.items || []) as Item[],
70	    total: Number(json.total || 0),
71	  };
72	}

74	async function fetchCategories() {
75	  try {
76	    const res = await authorizedFetch("/api/ecom-categories");
77	    if (!res.ok) return [];
78	    const json = await res.json();
79	    return (json?.items || []) as Array<{ slug: string; name: string; color?: string }>;
80	  } catch (error) {
81	    console.error("Failed to load categories", error);
82	    return [];
83	  }
84	}

498	function stringToHslColor(str: string, s = 60, l = 42) {
499	  let hash = 0;
500	  for (let i = 0; i < (str || "").length; i++) {
501	    hash = str.charCodeAt(i) + ((hash << 5) - hash);
502	  }
503	  const h = Math.abs(hash) % 360;
504	  return `hsl(${h} ${s}% ${l}%)`;
505	}

507	function CategoryBadge({
508	  slug,
509	  categories,
510	}: {
511	  slug: string | null;
512	  categories: Array<{ slug: string; name: string; color?: string }>;
513	}) {
514	  if (!slug) {
515	    return <span className="text-xs rounded px-2 py-0.5 border border-white/10">-</span>;
516	  }
517	  const category = categories.find((item) => item.slug === slug);
518	  const name = category?.name || slug;
519	  const color = category?.color || stringToHslColor(slug);
520	  return (
521	    <span
522	      className="inline-flex items-center gap-1 text-xs rounded px-2 py-0.5 border"
523	      style={{
524	        borderColor: color,
525	        background: `color-mix(in oklab, ${color} 15%, transparent)`,
526	      }}
527	    >
528	      <span className="inline-block w-2 h-2 rounded-full" style={{ background: color }} />
529	      {name}
530	    </span>
531	  );
532	}
